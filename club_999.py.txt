import json
import urllib.request
import ssl

class Club999Claimer:
    """Automation module for claiming gift codes on 999Club."""

    BASE_URL = "https://api.999club.com/api/v1/gift/claim"

    def __init__(self, timeout=10):
        self.timeout = timeout

    def claim_gift_code(self, account, gift_code):
        if not gift_code or not gift_code.strip():
            return False, "Gift code cannot be empty."

        token = account.get("token", "")
        credential = account.get("credential", "")

        headers = {
            "Content-Type": "application/json",
            "Authorization": f"Bearer {token}",
            "User-Agent": "Mozilla/5.0 (Linux; Android 12; Mobile) AppleWebKit/537.36",
            "Origin": "https://999club.com",
            "Referer": "https://999club.com/"
        }

        payload = json.dumps({
            "redemptionCode": gift_code.strip(),
            "account": credential
        }).encode("utf-8")

        ctx = ssl.create_default_context()
        ctx.check_hostname = False
        ctx.verify_mode = ssl.CERT_NONE

        req = urllib.request.Request(self.BASE_URL, data=payload, headers=headers, method="POST")

        try:
            with urllib.request.urlopen(req, timeout=self.timeout, context=ctx) as response:
                status = response.getcode()
                res_body = response.read().decode("utf-8")
                res_json = json.loads(res_body)

                if status in (200, 201) and res_json.get("code") in (0, 200, "0"):
                    msg = res_json.get("msg") or res_json.get("message") or "Gift code successfully redeemed!"
                    reward = res_json.get("data", {}).get("reward", "")
                    if reward:
                        msg += f" (+₹{reward})"
                    return True, msg
                else:
                    err_msg = res_json.get("msg") or res_json.get("message") or "Claim failed."
                    return False, f"999Club Error: {err_msg}"
        except urllib.error.HTTPError as e:
            try:
                err_json = json.loads(e.read().decode("utf-8"))
                err_msg = err_json.get("msg") or err_json.get("message") or str(e)
            except Exception:
                err_msg = f"HTTP Error {e.code}"
            return False, f"999Club HTTP {e.code}: {err_msg}"
        except Exception as e:
            return False, f"999Club Network Error: {str(e)}"
